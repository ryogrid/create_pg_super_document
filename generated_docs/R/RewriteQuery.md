## Overview
Recursively rewrites a query by applying rules and handling view updatability, forming the core of PostgreSQL's rule system processing.

## Definition
```c
static List *RewriteQuery(Query *parsetree, List *rewrite_events, int orig_rt_length)
```

## Detailed Description
RewriteQuery is the central function in PostgreSQL's query rewrite system that transforms queries by applying rules and handling automatic view updates. It operates recursively to handle complex rule scenarios and maintains protection against infinite recursion.

The function processes queries in several phases:
1. **CTE Processing**: Recursively processes DML statements in WITH clauses first
2. **Target List Rewriting**: Adjusts target lists for INSERT/UPDATE/MERGE operations using rewriteTargetListIU
3. **Rule Application**: Collects and fires appropriate rules using matchLocks and fireRules
4. **View Updatability**: For views without INSTEAD rules/triggers, attempts automatic view updates via rewriteTargetView
5. **Recursive Processing**: Recursively processes any product queries generated by rules
6. **Query Ordering**: Ensures proper execution order (INSERT queries first, UPDATE/DELETE queries last)

The function handles special cases including multi-row INSERT VALUES, ON CONFLICT clauses, MERGE operations, RETURNING clauses, and maintains proper recursion detection to prevent infinite rule loops.

## Parameters / Member Variables
- `parsetree`: The Query node to be rewritten
- `rewrite_events`: List of ongoing rewrite events for recursion detection
- `orig_rt_length`: Length of the original query's range table for product queries (0 for non-product queries)

## Notes and Other Information
- Maintains recursion detection through rewrite_events list to prevent infinite rule loops
- Handles command ordering: INSERT queries execute first, UPDATE/DELETE execute last to ensure rule actions see the right data
- For updatable views, transforms the query to target the base relation when no INSTEAD rules/triggers exist
- Supports complex scenarios like INSERT...VALUES with multiple rows, ON CONFLICT clauses, and MERGE operations
- Validates that WITH clauses don't generate multiple queries (which would break CTE semantics)
- Handles RETURNING clause validation ensuring INSTEAD rules provide appropriate RETURNING when needed
- The orig_rt_length parameter is used to skip already-processed VALUES RTEs in product queries generated by rules
- Throws specific errors for unsupported combinations like rules with ON CONFLICT or conditional rules in WITH clauses

static List *
RewriteQuery(Query *parsetree, List *rewrite_events, int orig_rt_length)
{
    CmdType event = parsetree->commandType;
    bool instead = false;
    bool returning = false;
    bool updatableview = false;
    Query *qual_product = NULL;
    List *rewritten = NIL;

    // First, recursively process any DML statements in WITH clauses
    foreach(lc1, parsetree->cteList)
    {
        CommonTableExpr *cte = lfirst_node(CommonTableExpr, lc1);
        Query *ctequery = castNode(Query, cte->ctequery);

        if (ctequery->commandType == CMD_SELECT)
            continue;

        List *newstuff = RewriteQuery(ctequery, rewrite_events, 0);

        // Handle rewrite results for WITH clauses (must be single query)
        if (list_length(newstuff) == 1)
        {
            ctequery = linitial_node(Query, newstuff);
            // Validate it's a supported command type
            if (!(ctequery->commandType == CMD_SELECT ||
                  ctequery->commandType == CMD_UPDATE ||
                  ctequery->commandType == CMD_INSERT ||
                  ctequery->commandType == CMD_DELETE ||
                  ctequery->commandType == CMD_MERGE))
            {
                ereport(ERROR, /* DO INSTEAD NOTIFY not supported */);
            }
            cte->ctequery = (Node *) ctequery;
        }
        else if (newstuff == NIL)
        {
            ereport(ERROR, /* DO INSTEAD NOTHING not supported */);
        }
        else
        {
            ereport(ERROR, /* Multi-statement rules not supported */);
        }
    }

    // Process non-SELECT commands (INSERT/UPDATE/DELETE/MERGE)
    if (event != CMD_SELECT && event != CMD_UTILITY)
    {
        int result_relation = parsetree->resultRelation;
        RangeTblEntry *rt_entry = rt_fetch(result_relation, parsetree->rtable);
        Relation rt_entry_relation = table_open(rt_entry->relid, NoLock);

        // Rewrite target lists based on command type
        if (event == CMD_INSERT)
        {
            // Handle INSERT (including multi-row VALUES)
            parsetree->targetList = rewriteTargetListIU(parsetree->targetList, ...);
            // Handle ON CONFLICT if present
            if (parsetree->onConflict &&
                parsetree->onConflict->action == ONCONFLICT_UPDATE)
            {
                parsetree->onConflict->onConflictSet =
                    rewriteTargetListIU(parsetree->onConflict->onConflictSet, ...);
            }
        }
        else if (event == CMD_UPDATE)
        {
            parsetree->targetList = rewriteTargetListIU(parsetree->targetList, ...);
        }
        else if (event == CMD_MERGE)
        {
            // Rewrite each MERGE action's target list
            foreach(lc1, parsetree->mergeActionList)
            {
                MergeAction *action = (MergeAction *) lfirst(lc1);
                if (action->commandType == CMD_UPDATE ||
                    action->commandType == CMD_INSERT)
                {
                    action->targetList = rewriteTargetListIU(action->targetList, ...);
                }
            }
        }

        // Apply rules to the query
        List *locks = matchLocks(event, rt_entry_relation, result_relation,
                                parsetree, &hasUpdate);
        List *product_queries = fireRules(parsetree, result_relation, event, locks,
                                         &instead, &returning, &qual_product);

        // Handle automatic view updates if no INSTEAD rules/triggers
        if (!instead &&
            rt_entry_relation->rd_rel->relkind == RELKIND_VIEW &&
            !view_has_instead_trigger(rt_entry_relation, event,
                                     parsetree->mergeActionList))
        {
            if (qual_product != NULL)
                error_view_not_updatable(rt_entry_relation, ...);

            // Attempt automatic view update
            parsetree = rewriteTargetView(parsetree, rt_entry_relation);

            // Add to product queries
            if (parsetree->commandType == CMD_INSERT)
                product_queries = lcons(parsetree, product_queries);
            else
                product_queries = lappend(product_queries, parsetree);

            instead = true;
            returning = true;
            updatableview = true;
        }

        // Recursively process product queries with recursion detection
        if (product_queries != NIL)
        {
            // Check for infinite recursion
            foreach(n, rewrite_events)
            {
                rewrite_event *rev = (rewrite_event *) lfirst(n);
                if (rev->relation == RelationGetRelid(rt_entry_relation) &&
                    rev->event == event)
                    ereport(ERROR, /* infinite recursion detected */);
            }

            // Add current event to recursion list
            rewrite_event *rev = (rewrite_event *) palloc(sizeof(rewrite_event));
            rev->relation = RelationGetRelid(rt_entry_relation);
            rev->event = event;
            rewrite_events = lappend(rewrite_events, rev);

            // Recursively rewrite each product query
            foreach(n, product_queries)
            {
                Query *pt = (Query *) lfirst(n);
                List *newstuff = RewriteQuery(pt, rewrite_events,
                                             pt == parsetree ?
                                             orig_rt_length :
                                             product_orig_rt_length);
                rewritten = list_concat(rewritten, newstuff);
            }

            rewrite_events = list_delete_last(rewrite_events);
        }

        // Validate RETURNING clauses with INSTEAD rules
        if ((instead || qual_product != NULL) &&
            parsetree->returningList && !returning)
        {
            ereport(ERROR, /* RETURNING not supported without proper rule */);
        }

        table_close(rt_entry_relation, NoLock);
    }

    // Add original query to results (ordering: INSERT first, others last)
    if (!instead)
    {
        if (parsetree->commandType == CMD_INSERT)
        {
            if (qual_product != NULL)
                rewritten = lcons(qual_product, rewritten);
            else
                rewritten = lcons(parsetree, rewritten);
        }
        else
        {
            if (qual_product != NULL)
                rewritten = lappend(rewritten, qual_product);
            else
                rewritten = lappend(rewritten, parsetree);
        }
    }

    // Validate CTE usage with multiple result queries
    if (parsetree->cteList != NIL)
    {
        int qcount = 0;
        foreach(lc1, rewritten)
        {
            Query *q = (Query *) lfirst(lc1);
            if (q->commandType != CMD_UTILITY)
                qcount++;
        }
        if (qcount > 1)
            ereport(ERROR, /* WITH not supported with multiple queries */);
    }

    return rewritten;
}
