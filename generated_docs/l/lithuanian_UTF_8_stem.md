## Detailed Description
This function implements the complete Lithuanian stemming algorithm using the Snowball stemming framework. It processes Lithuanian words through multiple sequential steps: 1) Sets up the R1 region by finding the first vowel-consonant-vowel sequence or ensuring minimum word length, 2) Applies conflict resolution for morphological ambiguities, 3) Performs primary suffix removal (step1), 4) Fixes 'ch' and 'dz' phonetic patterns, 5) Applies secondary suffix removal (step2), 6) Repeats phonetic fixes for 'ch'/'dz' and 'gd' patterns, and 7) Returns the cursor to the beginning for final result. The function uses backward processing (from end to beginning) for most operations and maintains state through cursor positions and region boundaries.

extern int lithuanian_UTF_8_stem(struct SN_env * z) {
    // Initialize R1 region to end of word
    z->I[0] = z->l;

    // Find R1 region: locate vowel-consonant-vowel pattern or use minimum length
    int start_pos = z->c;

    // Special handling for words starting with 'a' and longer than 6 UTF-8 chars
    if (z->c < z->l && z->p[z->c] == 'a' && len_utf8(z->p) > 6) {
        skip_utf8(z->p, z->c, z->l, 1);  // Skip first character
    }

    // Find first vowel, then consonant, then vowel to establish R1 boundary
    if (out_grouping_U(z, g_v, 97, 371, 1) >= 0) {  // Skip to first vowel
        z->c += out_grouping_U(z, g_v, 97, 371, 1);
        if (in_grouping_U(z, g_v, 97, 371, 1) >= 0) {  // Skip to next consonant
            z->c += in_grouping_U(z, g_v, 97, 371, 1);
            z->I[0] = z->c;  // Set R1 boundary
        }
    }

    z->c = start_pos;

    // Set up for backward processing (from end to beginning)
    z->lb = z->c;
    z->c = z->l;

    // Apply Lithuanian stemming steps in sequence
    r_fix_conflicts(z);     // Fix morphological conflicts
    r_step1(z);             // Remove primary suffixes
    r_fix_chdz(z);          // Fix ch/dz phonetic patterns
    r_step2(z);             // Remove secondary suffixes
    r_fix_chdz(z);          // Fix ch/dz patterns again
    r_fix_gd(z);            // Fix gd phonetic patterns

    // Return cursor to beginning
    z->c = z->lb;
    return 1;
}
