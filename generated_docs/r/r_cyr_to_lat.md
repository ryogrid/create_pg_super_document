    // Scan through entire string converting Cyrillic to Latin
    while(1) {
        int scan_pos = z->c;

        // Try to find a Cyrillic character at current position
        z->bra = z->c;

                case 3: slice_from_s(z, 1, s_2); break;   // в→v
